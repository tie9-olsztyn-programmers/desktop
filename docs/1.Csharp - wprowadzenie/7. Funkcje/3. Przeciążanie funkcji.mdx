# Przeciążanie funkcji (metod) w C# polega na
- definiowaniu kilku metod o tej samej nazwie
- w ramach jednej klasy, ale
- z różną liczbą, typami lub kolejnością parametrów.
- typ zwracany przez metodę nie ma wpływu na przeciążanie

```csharp
// Funkcja przyjmująca jeden argument typu int
public void Wyswietl(int liczba)
{
     Console.WriteLine($"Podano liczbę: {liczba}");
}

// Funkcja przyjmująca jeden argument typu string
public void Wyswietl(string tekst)
{
     Console.WriteLine($"Podano tekst: {tekst}");
}

// Funkcja przyjmująca jeden argument typu string, jeden argument typu int
public void Wyswietl(string tekst, int liczba)
{
     Console.WriteLine($"Podano tekst: {tekst} i liczbę {liczba}");
}

// Funkcja nic nie przyjmuje, ale zwraca string
public string Wyswietl()
{
    return "Ja tylko zwracam string";
}
```
