---
title: Pobieranie danych z konsoli
---


```csharp
using System;

public class Program
{
    public static void Main()
    {
        Console.WriteLine("Podaj swoje imię:");
        string imie = Console.ReadLine();  // Odczytuje dane od użytkownika jako string
        Console.WriteLine(string.Format("Cześć, {0}!", imie));

        Console.WriteLine("Podaj liczbę:");
        string input = Console.ReadLine();  // Odczytuje dane jako string
        int liczba = int.Parse(input);  // Konwertuje string na liczbę całkowitą
        Console.WriteLine(string.Format("Podałeś liczbę: {0}", liczba));
    }
}
```
