# Zapisywanie do pliku - SaveFileDialog

## Zapis do pliku `.txt`

Do projektu z poprzedniej lekcji dodajemy w `.axaml`:
```xml
<!-- .... -->

        <StackPanel Spacing="10" Margin="20">
            <Button Content="Zapisz do pliku" Click="OnSaveFileClick" />
            <TextBox Name="InputBox"
                     Height="200"
                     AcceptsReturn="True"
                     TextWrapping="Wrap"
                     />
        </StackPanel>

    </StackPanel>
```

W `.cs` musimy obsłużyć zapisu do pliku, więc dodajemy metodę `OnSaveFileClick`:
```cs
 private async void OnSaveFileClick(object? sender, RoutedEventArgs e)
    {
        try
        {

            var dialog = new SaveFileDialog
            {
                Title = "Zapisz plik tekstowy",
                InitialFileName = "dane.txt",
                DefaultExtension = "txt",
                Filters =
                {
                    new FileDialogFilter
                    {
                        Name = "Pliki tekstowe",
                        Extensions = { "txt" }
                    }
                }
            };

            string? path = await dialog.ShowAsync(this);

            if (path is null)
                return;

            await File.WriteAllTextAsync(path, InputBox.Text);
        }
        catch (Exception exception)
        {
           Console.WriteLine(exception.Message);
        }
    }
```

## Zapis do pliku `.json`
Do `.axaml` dodamy pole do wprowadzania tekstu i przycisk do zapisu do pliku `.json`:
```xml
<!-- .... -->
         <StackPanel Spacing="10" Margin="20">
            <Button Content="Zapisz JSON" Click="OnSaveJsonClick" />
            <TextBox Name="InputJsonBox" Height="100" AcceptsReturn="True" />
        </StackPanel>
    </StackPanel>

```


```cs
 private async void OnSaveJsonClick(object? sender, RoutedEventArgs e)
    {
        try
        {
            var dialog = new SaveFileDialog
            {
                Title = "Zapisz plik JSON",
                InitialFileName = "jsonData.json",
                DefaultExtension = "json",
                Filters =
                {
                    new FileDialogFilter
                    {
                        Name = "JSON",
                        Extensions = { "json" }
                    }
                }
            };


            string? path = await dialog.ShowAsync(this);

            if (path is null) return;

            var data = new
            {
                message = InputJsonBox.Text
            };

            // Konwertujemy obiekt na JSON string
            string json = JsonSerializer.Serialize(data, new JsonSerializerOptions
            {
                WriteIndented = true,
                Encoder = JavaScriptEncoder.UnsafeRelaxedJsonEscaping
            });


            await File.WriteAllTextAsync(path, json);
        }
        catch (Exception exception)
        {
            Console.WriteLine(exception.Message);
        }

    }
```
